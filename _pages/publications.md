<script>
function copyCitation(button) {
  const citation = button.dataset.citation;

  navigator.clipboard.writeText(citation).then(() => {
    button.textContent = "Copied!";
    setTimeout(() => {
      button.textContent = "Copy citation";
    }, 1200);
  });
}
</script>
